        private void Form1_Load(object sender, EventArgs e)

        private void PrazniKontrole()
        {
            textBox2.Clear();
            comboBox1.Text = "";
            maskedTextBox1.Clear();
            maskedTextBox2.Clear();
        }
        private void UnosAktivnosti()
        {
            if (string.IsNullOrWhiteSpace(textBox2.Text) ||
                string.IsNullOrWhiteSpace(comboBox1.Text) ||
                !maskedTextBox1.MaskCompleted ||
                !maskedTextBox2.MaskCompleted)
            {
                MessageBox.Show("Morate popuniti sva polja.");
                return;
            }

            Kon.Open();

            SqlCommand cmd = new SqlCommand("UnesiAktivnost", Kon);
            cmd.CommandType = CommandType.StoredProcedure;

            cmd.Parameters.Add("@NazivAktivnosti", SqlDbType.VarChar).Value = textBox2.Text;
            cmd.Parameters.Add("@Dan", SqlDbType.VarChar).Value = comboBox1.Text;
            cmd.Parameters.Add("@Pocetak", SqlDbType.Time).Value = TimeSpan.Parse(maskedTextBox1.Text);
            cmd.Parameters.Add("@Zavrsetak", SqlDbType.Time).Value = TimeSpan.Parse(maskedTextBox2.Text);

            cmd.ExecuteNonQuery();

            Kon.Close();
        }
        private void PuniKontroluLV()
        {
            listView1.Items.Clear();

            Kon.Open();

            SqlCommand cmd = new SqlCommand("PuniListView", Kon);
            cmd.CommandType = CommandType.StoredProcedure;

            dr = cmd.ExecuteReader();

            while (dr.Read())
            {

                ListViewItem red = new ListViewItem(dr[0].ToString());
                for (int i = 1; i < 5; i++) /* i IDE DO KOLIKO POLJA VRACA PROCEDURA*/
                    red.SubItems.Add(dr[i].ToString());
                listView1.Items.Add(red);
            }
            Kon.Close();
        }

        private void SaListViewNaKontrole()
        {
            if (listView1.SelectedItems.Count == 0)
                return;

            ListViewItem item = listView1.SelectedItems[0];

            textBox1.Text = item.SubItems[0].Text;        // ID / šifra
            textBox2.Text = item.SubItems[1].Text;        // naziv aktivnosti
            comboBox1.Text = item.SubItems[2].Text;       // dan
            maskedTextBox1.Text = item.SubItems[3].Text;  // početak
            maskedTextBox2.Text = item.SubItems[4].Text;  // završetak

            id = Convert.ToInt32(item.SubItems[0].Text);
        }
        private void button1_Click_1(object sender, EventArgs e)
        {
            UnosAktivnosti();
            PuniKontroluLV();
            PrazniKontrole();
        }
