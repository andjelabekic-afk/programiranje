 ALTER PROCEDURE [dbo].[PuniGridChart]
  AS
  SELECT CASE Aktivnosti.Dan
			WHEN 'Ponedeljak' THEN '1' + '-' + Aktivnosti.Dan
			WHEN 'Utorak' THEN '2' + '-' + Aktivnosti.Dan
			WHEN 'Sreda' THEN '3' + '-' + Aktivnosti.Dan
			WHEN 'Cetvrtak' THEN '4' + '-' + Aktivnosti.Dan
			WHEN 'Petak' THEN '5' + '-' + Aktivnosti.Dan
			WHEN 'Subota' THEN '6' + '-' + Aktivnosti.Dan
			WHEN 'Nedelja' THEN '7' + '-' + Aktivnosti.Dan
		END AS Dan1
        ,COUNT(DeteID) AS BrDece
  INTO TEMP#
  FROM Registar_Aktivnosti INNER JOIN Aktivnosti ON Registar_Aktivnosti.AktivnostID = Aktivnosti.AktivnostID 
  GROUP BY Aktivnosti.Dan
  SELECT SUBSTRING(Dan1, 3, 10) AS Dan 
       , BrDece
  FROM TEMP#
  ORDER BY Dan1
  DROP TABLE TEMP#
