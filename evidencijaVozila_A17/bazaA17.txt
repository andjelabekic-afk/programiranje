ALTER PROCEDURE [dbo].[AzurirajPodatkeVozilo] @Registracija as nvarchar(50)
                                      ,@GodinaProizvodnje as int
                                      ,@Kilometraza as int 
                                      ,@ModelNaziv as NVarChar(50)
									  ,@BojaNaziv as NVarChar(50)
									  ,@GorivoNaziv as NVarChar(50)
									  ,@Cena as money
									  ,@VoziloID as int
AS
UPDATE Vozilo
SET Registracija = @Registracija
   ,GodinaProizvodnje = @GodinaProizvodnje
   ,PredjenoKM = @Kilometraza
   ,ModelID = (SELECT ModelID FROM Model WHERE Naziv = @ModelNaziv)
   ,BojaID = (SELECT BojaID FROM Boja WHERE Naziv = @BojaNaziv)
   ,GorivoID = (SELECT GorivoID FROM Gorivo WHERE Naziv = @GorivoNaziv)
   ,Cena = @Cena
WHERE VoziloID = @VoziloID

ALTER PROCEDURE [dbo].[PuniGridChart] @Model AS NVarChar(50)
AS
SELECT GodinaProizvodnje
      ,round(AVG(Cena),2) aS ProsCena
  FROM Vozilo INNER JOIN Model ON Vozilo.ModelID = Model.ModelID
  WHERE Model.Naziv = @Model
  GROUP BY GodinaProizvodnje

ALTER PROCEDURE [dbo].[PuniListBoxView]
AS
SELECT VoziloID
      ,Registracija
	  ,GodinaProizvodnje
	  ,PredjenoKM
	  ,Vozilo.ModelID
	  ,Vozilo.BojaID
	  ,Vozilo.GorivoID
	  ,Cena
	  ,Model.Naziv as Model
	  ,Boja.Naziv as Boja
	  ,Gorivo.Naziv as Gorivo
FROM Vozilo LEFT JOIN Model ON Vozilo.ModelID = Model.ModelID
            LEFT JOIN Boja ON Vozilo.BojaID = Boja.BojaID
			LEFT JOIN Gorivo ON Vozilo.GorivoID = Gorivo.GorivoID
