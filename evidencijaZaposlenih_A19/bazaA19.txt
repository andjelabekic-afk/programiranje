ALTER PROCEDURE [dbo].[PuniChart]
AS
SELECT [Ime]
      ,[Prezime]
	  ,DateDiff(DAY, RukovodiSektorom.DatumPostavljanja, IsNull(RukovodiSektorom.DatumRazresenja, GETDATE())) AS Dana
FROM Radnik INNER JOIN Sektor ON Radnik.SektorID = Sektor.SektorID
            INNER JOIN RukovodiSektorom ON Radnik.RadnikID = RukovodiSektorom.RadnikID 
WHERE Sektor.Naziv = 'Finansijski sektor'
ORDER BY Radnik.RadnikID

/****** Script for SelectTopNRows command from SSMS  ******/
ALTER PROCEDURE [dbo].[PuniListView]
@SektorNaziv AS NVarChar(50)
AS
SELECT Radnik.RadnikID
      ,[Ime]
      ,[Prezime]
	  ,RukovodiSektorom.DatumPostavljanja
	  ,IsNull(CAST(RukovodiSektorom.DatumRazresenja AS varchar) , 'Do danas') AS DatumRazresenja
      ,Radnik.SektorID
	  ,Sektor.Naziv
  FROM Radnik INNER JOIN Sektor ON Radnik.SektorID = Sektor.SektorID
              INNER JOIN RukovodiSektorom ON Radnik.RadnikID = RukovodiSektorom.RadnikID 
  WHERE Sektor.Naziv = @SektorNaziv
  ORDER BY RukovodiSektorom.DatumPostavljanja, RukovodiSektorom.DatumRazresenja
