ALTER PROCEDURE [dbo].[PuniGridChart] @Period AS Int
AS
SELECT YEAR(DatumPocetka) AS Godina
     , (Projekat.ProjekatID) AS BrojProjekata
	 , COUNT(Angazman.RadnikID) AS BrRadnika
INTO #TEMP
FROM Projekat INNER JOIN Angazman ON Projekat.ProjekatID = Angazman.ProjekatID
WHERE Year(DatumPocetka) > (YEAR(Getdate()) - @Period)
GROUP BY YEAR(DatumPocetka), Projekat.ProjekatID
ORDER BY YEAR(DatumPocetka)

SELECT Godina
     , COUNT(BrojProjekata) AS BrojProjekata 
	 , SUM(BrRadnika) AS BrRadnika
FROM #TEMP
GROUP BY Godina
DROP TABLE #TEMP
