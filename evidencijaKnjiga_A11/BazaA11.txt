ALTER PROCEDURE [dbo].[PuniChart]  @Autor1 AS NVarChar(50), @Autor2 AS NVarChar(50), @Autor3 AS NVarChar(50)
AS
SELECT COUNT(Knjiga.KnjigaID) AS BrKnjiga
       , Rtrim(Autor.Prezime) + ' ' + Rtrim(Autor.Ime) AS Autor
FROM Knjiga INNER JOIN Autor_Izdanje ON Knjiga.KnjigaID = Autor_Izdanje.KnjigaID
            INNER JOIN Autor ON Autor_Izdanje.AutorID = Autor.AutorID
WHERE Rtrim(Autor.Prezime) + ' ' + Rtrim(Autor.Ime) = @Autor1 OR
      Rtrim(Autor.Prezime) + ' ' + Rtrim(Autor.Ime) = @Autor2 OR
      Rtrim(Autor.Prezime) + ' ' + Rtrim(Autor.Ime) = @Autor3
GROUP BY Rtrim(Autor.Prezime),Rtrim(Autor.Ime)
