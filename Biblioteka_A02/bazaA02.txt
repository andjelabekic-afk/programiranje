ALTER PROCEDURE [dbo].[PuniGridChart] @SifraAutora as Int , @Period As Int
AS
SELECT  Year(DatumUzimanja) as GodUzimanja
	  , COUNT(*) AS BrIznajmljivanja
FROM NaCitanju INNER JOIN Napisali ON NaCitanju.KnjigaID = Napisali.KnjigaID
               INNER JOIN Autor ON Napisali.AutorID = Autor.AutorID 
WHERE Autor.AutorID = @SifraAutora AND Year(DatumUzimanja) >= (YEAR(Getdate()) - @Period)
GROUP BY Year(DatumUzimanja)
